{{- /* Icon tab strip for the momo 2027 page. Each child {{< sectiontab >}} becomes one tab plus one panel. Every panel is rendered into the HTML so the page still reads without JavaScript and stays searchable; js/core/section-tabs.js hides the inactive ones and keeps the URL hash in sync. @example {{< sectiontabs >}} {{< sectiontab title="Overview" icon="1-Overview.png" >}}text{{< /sectiontab >}} {{< /sectiontabs >}} */ -}} {{- /* Unused, but required to populate the store from the child shortcodes. */ -}} {{- .Inner -}} {{- $sections := .Store.Get "sections" | default slice -}} {{- if eq (len $sections) 0 -}} {{- errorf "sectiontabs must contain at least one sectiontab (%s)" .Position -}} {{- end -}} {{- $page := .Page -}}
{{- range $i, $s := $sections -}} {{- $img := "" -}} {{- with $s.icon -}} {{- with resources.Get (printf "images/2027/%s" .) -}} {{- $img = .Process "resize 96x png" -}} {{- else -}} {{- errorf "sectiontab %q: icon not found at assets/images/2027/%s" $s.title . -}} {{- end -}} {{- end -}} {{- with $img -}} {{- end -}} {{ $s.title }} {{- end -}}
{{- range $i, $s := $sections -}}
{{- $s.content | $page.RenderString (dict "display" "block") -}}
{{- end -}}
